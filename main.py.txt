import os
import glob
import telebot
import google.generativeai as genai
from PIL import Image

# جلب المفاتيح بأمان من بيئة الاستضافة
TELEGRAM_TOKEN = os.getenv("TELEGRAM_TOKEN")
GEMINI_API_KEY = os.getenv("GEMINI_API_KEY")

genai.configure(api_key=GEMINI_API_KEY)

# قراءة جميع الملفات الموجودة في مجلد knowledge تلقائياً (مهما كان عددها)
knowledge_text = ""
knowledge_files = glob.glob("knowledge/*.txt") + glob.glob("knowledge/*.md")

for file_path in knowledge_files:
    try:
        with open(file_path, "r", encoding="utf-8") as f:
            knowledge_text += f"\n\n--- [مصدر من ملف: {os.path.basename(file_path)}] ---\n" + f.read()
    except Exception as e:
        print(f"خطأ في قراءة الملف {file_path}: {e}")

# تعليمات النظام الصارمة لتوجيه البوت للاعتماد على ملفاتك حصرياً في مجال كرافتلاند
system_instruction = f"""
أنت مساعد ذكي وخبير محترف في لعبة فري فاير كرافتلاند (Free Fire Craftland)، وتستند في إجاباتك على الملفات الضخمة التي زودتك بها.

⚠️ قواعد التشغيل الأساسية:
1. قاعدة المعرفة الخاصة بك (المرجع الأساسي):
{knowledge_text}

2. يجب عليك دائماً مراجعة والبحث في هذه الملفات أولاً قبل الإجابة على أي استفسار يخص كرافتلاند، الأكواد، السكربتات، أو أنظمة العمل الموجودة فيها.
3. مسموح لك تحليل الصور المرسلة (مثل لقطات الشاشة لأخطاء اللعبة أو الأكواد)
4. إذا سأل المستخدم عن مواضيع خارج نطاق كرافتلاند، أجب بمرونة وبشكل طبيعي،
"""

# إعداد نموذج جيميناي
model = genai.GenerativeModel(
    model_name="gemini-1.5-flash",
    system_instruction=system_instruction
)

bot = telebot.TeleBot(TELEGRAM_TOKEN)

@bot.message_handler(commands=['start', 'help'])
def send_welcome(message):
    bot.reply_to(message, "أهلاً بك! ا🗺️🤖\nأنا مبرمج لقراءة جميع ملفات الشروحات وقواعد البيانات الخاصة بك، وجاهز لتحليل الصور والإجابة على أي استفسار بدقة فوراً.")

# استقبال النصوص والصور معاً وتحليلها
@bot.message_handler(content_types=['text', 'photo'])
def handle_messages(message):
    try:
        contents = []
        user_text = message.caption if message.caption else message.text
        
        if not user_text and message.photo:
            user_text = "حلل هذه الصورة المتعلقة بكرافتلاند بناءً على ملفات المعرفة المتاحة لديك."

        # معالجة الصور المرفقة
        if message.photo:
            file_info = bot.get_file(message.photo[-1].file_id)
            downloaded_file = bot.download_file(file_info.file_path)
            image_path = "temp_user_image.jpg"
            
            with open(image_path, "wb") as new_file:
                new_file.write(downloaded_file)
            
            pil_image = Image.open(image_path)
            contents.append(pil_image)

        if user_text:
            contents.append(user_text)

        # توليد الرد من جيميناي بناءً على الملفات والصورة/النص
        response = model.generate_content(contents)
        bot.reply_to(message, response.text)

    except Exception as e:
        bot.reply_to(message, f"عذراً، حدث خطأ أثناء معالجة الطلب: {str(e)}")
